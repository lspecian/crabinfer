    pub(crate) qzeros: Tensor,    // [num_groups, out_features / 8] u32

    pub(crate) group_size: usize,

}

impl AwqLinear {
    pub fn from_parts(qweight: Tensor, scales: Tensor, qzeros: Tensor, bias: Option<Tensor>, group_size: usize) -> Result<Self>;

  <files>crabinfer-core/src/serving/safetensors_loader.rs</files>
  <behavior>
    - GptqConfig parses: {"bits":4, "group_size":128, "desc_act":false, "sym":true, "quant_method":"gptq"}
    - GptqConfig defaults desc_act to false when missing
    - AwqConfig parses from quantize_config.json: {"quant_method":"awq", "q_group_size":128, "w_bit":4, "version":"GEMM"}
    - AwqConfig parses from config.json embedded: {"quantization_config": {"quant_method":"awq", ...}}
    - detect_quant_method reads quantize_config.json and returns Gptq or Awq
    - detect_quant_method falls back to config.json quantization_config if no quantize_config.json
    - detect_quant_method returns None for non-quantized models
    - desc_act=true models produce a clear error
    - load_gptq_linear correctly maps weight prefix to qweight/qzeros/scales tensors
  </behavior>
  <action>
    Add the following to `safetensors_loader.rs`:

    1. **Quant config structs** (after the existing HfConfig):
       ```rust
       #[derive(Debug, serde::Deserialize)]
       struct GptqConfig {
           bits: usize,
           group_size: usize,
           #[serde(default)]
           desc_act: bool,
           #[serde(default = "default_quant_method_gptq")]
           quant_method: String,
       }

       #[derive(Debug, serde::Deserialize)]
       struct AwqConfig {
           #[serde(alias = "w_bit")]
           bits: usize,
           #[serde(alias = "q_group_size")]
           group_size: usize,
           #[serde(default)]
           version: Option<String>,
       }

       #[derive(Debug, serde::Deserialize)]
       struct HfConfigQuantization {
           #[serde(default)]
           quantization_config: Option<serde_json::Value>,
       }
       ```

    2. **detect_quant_config()** function:
       - Read `quantize_config.json` if it exists in model_dir
       - Parse the `quant_method` field: "gptq" -> try deserialize as GptqConfig, "awq" -> AwqConfig
       - If no `quantize_config.json`, check `config.json` for `quantization_config` key
       - Return `Option<(QuantizationMethod, usize /* group_size */, usize /* bits */)>`
       - If `desc_act == true`, return an error: "GPTQ models with desc_act=true are not yet supported. Use a model quantized with desc_act=false."

    3. **load_gptq_linear()** function:
       ```rust
       fn load_gptq_linear(
           weights: &HashMap<String, Tensor>,
           prefix: &str,  // e.g. "model.layers.0.self_attn.q_proj"
           group_size: usize,
           device: &Device,
       ) -> Result<MaybeQuantizedLinear>
       ```
       - Load `{prefix}.qweight`, `{prefix}.qzeros`, `{prefix}.scales` from weights map
       - Load optional `{prefix}.bias`
       - Validate qweight shape: if `qweight.dim(1) > qweight.dim(0) * 8`, the weights are stored transposed (column-major). Transpose to the expected [in_features/8, out_features] layout. This ensures models from any HuggingFace publisher work regardless of their storage convention.
       - Call `GptqLinear::from_parts(qweight, scales, qzeros, bias, group_size)` and wrap in `MaybeQuantizedLinear::Gptq`

    4. **load_awq_linear()** — same as load_gptq_linear but wraps in `MaybeQuantizedLinear::Awq` via `AwqLinear::from_parts()`

    5. **Modify load_model_from_safetensors():**
       - After loading HfConfig, call `detect_quant_config(model_dir)` to auto-detect quantization
       - If quantization is detected AND the caller passed `QuantizationMethod::Gptq` or `QuantizationMethod::Awq` (or `None` meaning auto-detect), use the detected config
       - When loading layers, if GPTQ/AWQ detected, use `load_gptq_linear`/`load_awq_linear` instead of `load_linear` for projection weights (q_proj, k_proj, v_proj, o_proj, gate_proj, down_proj, up_proj)
       - Norm weights and embeddings are always loaded as FP tensors (not quantized)
       - Log: "Detected GPTQ quantization: bits={bits}, group_size={group_size}"

    6. **Unit tests** in the existing `#[cfg(test)] mod tests`:
       - `test_gptq_config_parse` — deserialize sample GPTQ quantize_config.json
       - `test_awq_config_parse` — deserialize AWQ config from both standalone and embedded formats
       - `test_gptq_config_desc_act_error` — verify desc_act=true produces error
       - `test_detect_quant_method` — create temp dirs with different config files, verify detection

    IMPORTANT per RESEARCH.md:
    - AWQ uses `q_group_size` (not `group_size`) and `w_bit` (not `bits`) — use serde `alias` attribute
    - Some AWQ models embed quant config in config.json under `quantization_config`, not in quantize_config.json — check both
    - Do NOT load g_idx tensor; error on desc_act=true models
    - Load qweight/qzeros/scales to device without dtype conversion (they are u32/f16 as-is)
